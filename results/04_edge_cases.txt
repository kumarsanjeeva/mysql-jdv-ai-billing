SET @doc = (SELECT data FROM order_dv WHERE JSON_VALUE(data, '$.request_id') = 'seed-order-002');

    JSON_UNQUOTE(JSON_SEARCH(@doc, 'one', 'seed-payment-001', NULL, '$.payments[*].request_id')),
